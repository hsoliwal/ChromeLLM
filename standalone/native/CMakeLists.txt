cmake_minimum_required(VERSION 3.20)
project(chromellm_llama_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(JNI REQUIRED)
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

option(CHROMELLM_JNI_CONTRACT_FIXTURE "Build synthetic JNI contract fixture (NEVER ship)" OFF)
option(CHROMELLM_JNI_ABI_MISMATCH_FIXTURE "Build incompatible synthetic ABI fixture (NEVER ship)" OFF)
if(CHROMELLM_JNI_ABI_MISMATCH_FIXTURE AND NOT CHROMELLM_JNI_CONTRACT_FIXTURE)
  message(FATAL_ERROR "ABI mismatch fixture requires CHROMELLM_JNI_CONTRACT_FIXTURE=ON")
endif()
set(LLAMA_CPP_SOURCE_DIR "" CACHE PATH "Exact, locally checked out llama.cpp source")
if(NOT CHROMELLM_JNI_CONTRACT_FIXTURE)
  if(NOT EXISTS "${LLAMA_CPP_SOURCE_DIR}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_CPP_SOURCE_DIR to an independently licensed, pinned llama.cpp checkout")
  endif()
  set(CHROMELLM_LLAMA_COMMIT "889f2ec96deacbc2d8a9a67dbe181eaa43a2fb3b")
  execute_process(COMMAND git -C "${LLAMA_CPP_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE LLAMA_REV OUTPUT_STRIP_TRAILING_WHITESPACE
    RESULT_VARIABLE REV_RESULT ERROR_QUIET)
  if(NOT REV_RESULT EQUAL 0 OR NOT LLAMA_REV STREQUAL CHROMELLM_LLAMA_COMMIT)
    message(FATAL_ERROR "llama.cpp checkout must be exactly ${CHROMELLM_LLAMA_COMMIT}")
  endif()
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  add_subdirectory("${LLAMA_CPP_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama-cpp")
endif()

add_library(chromellm_llama_jni SHARED chromellm_llama_jni.cpp)
target_include_directories(chromellm_llama_jni PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(chromellm_llama_jni PRIVATE Threads::Threads)
if(CHROMELLM_JNI_CONTRACT_FIXTURE)
  target_compile_definitions(chromellm_llama_jni PRIVATE CHROMELLM_JNI_CONTRACT_FIXTURE=1)
  if(CHROMELLM_JNI_ABI_MISMATCH_FIXTURE)
    target_compile_definitions(chromellm_llama_jni PRIVATE CHROMELLM_JNI_ABI_MISMATCH_FIXTURE=1)
    set_target_properties(chromellm_llama_jni PROPERTIES OUTPUT_NAME chromellm_llama_jni_abi_mismatch_fixture)
  else()
    set_target_properties(chromellm_llama_jni PROPERTIES OUTPUT_NAME chromellm_llama_jni_fixture)
  endif()
else()
  target_include_directories(chromellm_llama_jni PRIVATE "${LLAMA_CPP_SOURCE_DIR}/include")
  target_link_libraries(chromellm_llama_jni PRIVATE llama)
endif()
